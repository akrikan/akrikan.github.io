\hypertarget{introduction}{%
\section{Introduction}\label{introduction}}

This comprehensive reference covers all MathJax LaTeX commands and
syntax available in Quarto documents. MathJax supports most standard
LaTeX mathematical notation with some limitations.

\hypertarget{basic-math-modes}{%
\section{Basic Math Modes}\label{basic-math-modes}}

\hypertarget{inline-math}{%
\subsection{Inline Math}\label{inline-math}}

Use single dollar signs for inline math: \texttt{\$x\ =\ y\ +\ z\$}
renders as \(x = y + z\)

\hypertarget{display-math}{%
\subsection{Display Math}\label{display-math}}

Use double dollar signs for display math:

\begin{verbatim}
$$E = mc^2$$
\end{verbatim}

\[E = mc^2\]

\hypertarget{equation-environment}{%
\subsection{Equation Environment}\label{equation-environment}}

\begin{verbatim}
$$
\begin{equation}
f(x) = ax^2 + bx + c
\end{equation}
$$
\end{verbatim}

\[
\begin{equation}
f(x) = ax^2 + bx + c
\end{equation}
\]

\end{align}\]

\hypertarget{performance-tips}{%
\subsection{Performance Tips}\label{performance-tips}}

\begin{itemize}
\tightlist
\item
  Use \texttt{\$\$} for display math instead of
  \texttt{\textbackslash{}{[}} and \texttt{\textbackslash{}{]}}
\item
  Avoid unnecessary \texttt{\textbackslash{}left} and
  \texttt{\textbackslash{}right} when fixed-size delimiters work
\item
  Use \texttt{\textbackslash{}text\{\}} for text within math expressions
\item
  Break long expressions into multiple lines using \texttt{align}
\end{itemize}

\hypertarget{common-mistakes}{%
\subsection{Common Mistakes}\label{common-mistakes}}

\begin{itemize}
\tightlist
\item
  Forgetting to escape special characters: \texttt{\textbackslash{}\{},
  \texttt{\textbackslash{}\}}, \texttt{\textbackslash{}\&},
  \texttt{\textbackslash{}\%}
\item
  Using \texttt{*} for multiplication instead of
  \texttt{\textbackslash{}cdot} or \texttt{\textbackslash{}times}
\item
  Not using proper spacing in integrals:
  \texttt{\textbackslash{}int\ f(x)\ \textbackslash{},\ dx}
\item
  Mixing text and math modes incorrectly
\end{itemize}
